# The numbers the game plays by that no card, prop or powerup sets. Each [section] holds `key number` lines;
# a mod's tuning.txt only needs the lines it changes. Distances are in pixels (a tile is 16), times in seconds,
# angles in radians.

[player]
hp 100          # health before upgrades
speed 48        # pixels a second before upgrades
magnet 28       # how near money and health drift in from, before upgrades
crit 0.1        # chance a hit lands double, before upgrades
pickup 7        # how near the player picks something up
width 8         # the player's footprint for walls, in pixels
depth 6
turn 2.6        # radians a second the view turns with a turn button held

[night]
length 600      # seconds from dusk to dawn
dusk_near 200   # zombies rise between these distances from the player at dusk,
dusk_far 250
dawn_near 56    # closing to these by dawn
dawn_far 110
home 12         # how near the farmhouse's front step counts as home once dawn breaks
rare 0.12       # chance a level-up card is rare and counts twice
push 80         # how far a level-up pushes zombies back

[easy]
damage 0.55     # multiplies the damage zombies deal
hurt_time 0.7   # seconds a hit leaves the player safe from the next
health_drop 60  # one kill in this many drops health; the rest drop money
heal 30         # health a health drop gives back

[normal]
damage 1
hurt_time 0.5
health_drop 200
heal 20

[hard]
damage 1.4
hurt_time 0.4
health_drop 400
heal 15

[zombies]
max 300         # most alive at once
rise 0.9        # seconds a zombie takes to climb out of the ground, harmless and rooted
stagger 0.4     # seconds a heavy hit stops a zombie
growth 0.08     # health added per minute of the night, as a fraction of its own
leader_hp 4     # times a big zombie's health a horde's leader has
fury 1.01       # after dawn, every zombie's health and bite multiply by this each second
width 6         # a zombie's footprint for walls, in pixels
depth 4
walk_fps 8      # walking frames a second for the player, and for a zombie at a speed of 20
bite_reach 9    # how near a zombie bites the player
charge_reach 48 # how near a zombie with a charge speeds up
chew 0.5        # bites a second a zombie takes out of a prop in its way

[harvester]
hp 2500         # dawn's boss
speed 40
bite 2          # times a zombie's bite
rise 90         # how far from the player it rises, on the side away from home

[grenade]
recharge 12     # seconds to earn a grenade back, before upgrades
flight 0.7      # seconds in the air
radius 36
damage 50
bomblet_radius 18
bomblet_damage 25
aim_window 3    # single screen: real seconds to aim before it throws itself
aim_slow 0.5    # game speed while aiming
aim_speed 160   # cursor, map pixels a second
aim_memory 5    # seconds a throw's angle off the view and distance carry to the next aim
aim_ahead 64    # how far straight ahead the cursor starts otherwise

[barrel]
hp 20
radius 40
damage 60

[weapons]
view_half 0.6   # half the angle ahead the player sees and aims into, in radians, before a lantern widens it
behind_half 0.9 # half the angle behind a holster fires into, in radians
point_blank 70  # the pistol turns on anything this close, whichever way the player faces
crow_reach 7    # how near a crow pecks
dog_reach 7     # how near a dog bites
dog_stray 48    # a dog with nothing to chase runs back once this far from the player
dog_trot 0.45   # a dog's pace near the player, as a fraction of its speed
fire_burn 1.5   # seconds fire and flares keep a zombie burning after it leaves them

[blast]
hurt 15         # damage any blast deals the player, less armor
near 0.7        # the player is hurt within this fraction of its radius
