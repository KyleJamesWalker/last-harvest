# The numbers the game plays by that no card, prop or powerup sets. Each [section] holds `key number` lines;
# a mod's tuning.txt only needs the lines it changes. Distances are in pixels (a tile is 16), times in seconds,
# angles in radians.

[player]
hp 100          # health before upgrades
speed 48        # pixels a second before upgrades
magnet 28       # how near hearts and health drift in from, before upgrades
crit 0.1        # chance a hit lands double, before upgrades
pickup 7        # how near the player picks something up
width 8         # the player's footprint for walls, in pixels
depth 6
turn 2.6        # radians a second the view turns with a turn button held

[night]
length 600      # seconds from dusk to dawn
dusk_near 200   # zombies rise between these distances from the player at dusk,
dusk_far 250

dawn_far 110
home 12         # how near the farmhouse's front step counts as home once dawn breaks
rare 0.12       # chance a level-up card is rare and counts twice
push 80         # how far a level-up pushes zombies back

[levelup]
held_weight 1   # a card for a powerup the player holds is this many times as likely as one for a new powerup
banish 2        # cards a run can banish: gone for the rest of the run, its place dealt again
reroll 2        # times a run can deal all three cards again
favor 1         # cards a run can favor: taken, and this many times as likely for the rest of the run...
favor_weight 4  # ...as favor_weight says, on top of held_weight
favor_level 6   # the level from which favor is offered
breaker 2       # a rare card one level below the top breaks one past it, adding this many of its last step; 0 is off
